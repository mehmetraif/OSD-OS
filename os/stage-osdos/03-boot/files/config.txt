# OSD/OS image boot configuration. More options: http://rptl.io/configtxt
#
# The display output is set in osdos-display.txt, next to this file, a copy of
# one of the presets beside it. Settings → Display Output switches between the
# ones the Pi has; from any computer, copy one over it:
#   osdos-display-hdmi.txt            HDMI, resolution auto-detected
#   osdos-display-crt-ntsc.txt        composite, NTSC, 4:3 (AV jack, Pi 5 TV pads)
#   osdos-display-crt-pal.txt         composite, PAL, 4:3
#   osdos-display-crt-gpio-ntsc.txt   composite on a Pi 5's GPIO pins, NTSC
#   osdos-display-crt-gpio-pal.txt    composite on a Pi 5's GPIO pins, PAL
#   osdos-display-scart-rgb-ntsc.txt  RGB on the GPIO pins for SCART, 480i
#   osdos-display-scart-rgb-pal.txt   RGB on the GPIO pins for SCART, 576i
#   osdos-display-scart-rgb-240p.txt  the same on a Pi 4, 240p
#   osdos-display-scart-rgb-288p.txt  the same on a Pi 4, 288p
#   osdos-display-jamma-240p.txt      a JAMMA cabinet through a Raspberry Pi to
#                                     JAMMA board (RGB-Pi's layout), 240p: its
#                                     controls too, and on a Pi 4 its sound
#   osdos-display-jamma-240p-native.txt, -240p-super.txt, -288p.txt and
#   -288p-super.txt                   the same board at 320x240, 1920x240,
#                                     720x288 or 1920x288
#   osdos-display-jamma-480i.txt and -576i.txt
#                                     the same on a Pi 4, interlaced
#   osdos-display-jamma-25k.txt and -31k.txt
#                                     the same for a 25 kHz or 31 kHz (VGA)
#                                     monitor ONLY: too high a scan rate can
#                                     damage a 15 kHz one
# On a Pi 5 a SCART RGB preset also needs its composite sync (copied by hand,
# /etc/modprobe.d/osdos-display.conf: options drm_rp1_dpi force_csync=1), and
# the JAMMA ones must not have it (delete that file).
#
# The per-model settings below are the ones INSTALL.md documents.

# --- Global ---
auto_initramfs=1
disable_splash=1
disable_overscan=1
dtparam=audio=on
arm_64bit=1
# No pause before loading the kernel (the firmware waits a second by default).
boot_delay=0

include osdos-display.txt

# --- Pi 3B ---
[pi3]

# Drivers & Video
dtoverlay=vc4-fkms-v3d,cma-256

# Overclocking
over_voltage=4
arm_freq=1300
core_freq=450
sdram_freq=500

# --- Pi 3B+ ---
[pi3+]

# Drivers & Video
dtoverlay=vc4-fkms-v3d,cma-256

# Overclocking
over_voltage=2
arm_freq=1500
core_freq=500
sdram_freq=500

# --- Pi 4B ---
[pi4]

# Drivers & Video
dtoverlay=vc4-fkms-v3d,cma-256
dtoverlay=rpivid-v4l2

# Overclocking
over_voltage=2
arm_freq=1750
gpu_freq=600

# --- Pi 5 ---
# Its video overlay depends on the output, so it is set in osdos-display.txt.

# --- Global ---
[all]
